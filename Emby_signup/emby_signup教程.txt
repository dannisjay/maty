1.创建文件/opt/emby_signup/index.php

2.安装apache2和php

sudo apt install apache2 php libapache2-mod-php -y

3.修改/etc/apache2/ports.conf

4.修改/etc/apache2/sites-available/000-default.conf

5.设置文件夹权限

sudo chown -R www-data:www-data /opt/emby_signup

sudo chmod -R 755 /opt/emby_signup

6.测试配置并重启
# 检查配置语法
sudo apache2ctl configtest

# 重启Apache
sudo systemctl restart apache2

# 检查状态
sudo systemctl status apache2
